\begin{frame}{4. Evid\^encia espectral: MSC}
\large
Para cada \`epoca $m$, a FFT fornece o coeficiente complexo $X_m(f_0)$ na
frequ\^encia de interesse $f_0$.

\begin{block}{Magnitude quadr\'atica da coer\^encia entre \`epocas}
\[
\mathrm{MSC}(f_0)=
\frac{\left|\displaystyle\sum_{m=1}^{M}X_m(f_0)\right|^2}
     {M\displaystyle\sum_{m=1}^{M}|X_m(f_0)|^2}
\]
\end{block}

\begin{columns}[T]
\column{0.48\textwidth}
\centering
\textbf{Fases dispersas}\\[0.15cm]
\begin{tikzpicture}[scale=0.75]
  \draw[->] (-1.3,0) -- (1.3,0);
  \draw[->] (0,-1.3) -- (0,1.3);
  \foreach \a in {20,105,185,265,320} {
    \draw[-{Latex[length=2mm]},colControl,thick] (0,0) -- (\a:1.05);
  }
\end{tikzpicture}\\[-0.1cm]
MSC tende a ser menor

\column{0.48\textwidth}
\centering
\textbf{Fases alinhadas}\\[0.15cm]
\begin{tikzpicture}[scale=0.75]
  \draw[->] (-1.3,0) -- (1.3,0);
  \draw[->] (0,-1.3) -- (0,1.3);
  \foreach \a in {8,18,25,31,40} {
    \draw[-{Latex[length=2mm]},colPresent!55!black,thick]
      (0,0) -- (\a:1.05);
  }
\end{tikzpicture}\\[-0.1cm]
MSC tende a ser maior
\end{columns}

\note{A MSC combina magnitude e consist\^encia de fase entre as \`epocas. Ela
n\~ao \`e CSM: na CSM, cada coeficiente seria normalizado para m\'odulo unit\'ario
antes da soma.}
\end{frame}

% ====================================================================
\begin{frame}{O teste espectral n\~ao \`e o ponto central}
\large

\begin{block}{A MSC \`e uma escolha, n\~ao uma restri\c{c}\~ao do modelo}
O HMM precisa de uma sequ\^encia de evid\^encias ao longo do tempo, mas essa
evid\^encia n\~ao precisa vir exclusivamente da MSC.
\end{block}

\begin{itemize}
  \item Podemos trocar a MSC por outro teste que seja mais adequado aos dados;
  \item tamb\'em podemos combinar testes complementares, se isso produzir uma
  evid\^encia mais robusta;

\end{itemize}

\vfill
\centering
\textbf{Detector espectral} $\longrightarrow$ \textbf{evid\^encia observada}
$\longrightarrow$ \textbf{HMM}
\end{frame}

% ====================================================================
\begin{frame}{5. ORD $\rightarrow$ Observa\c{c}\~ao}
\large
Sob a hip\'otese de aus\^encia de resposta, a MSC possui uma distribui\c{c}\~ao
nula conhecida para uma janela com $M$ \`epocas:
\[
  \mathrm{MSC}\mid H_0 \sim \mathrm{Beta}(1,M-1).
\]
O valor de MSC \`e convertido em um $p$-valor e depois em um s\'imbolo ordinal:

\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \foreach \x/\w/\cor/\rotulo in {
    0/2.3/colAbsent/muito baixo,
    2.3/2.3/colSignal!35/baixo,
    4.6/2.3/colEvidence!45/m\'edio,
    6.9/2.3/colEvidence!75/alto,
    9.2/2.3/colControl!75/muito alto} {
    \filldraw[fill=\cor,draw=white] (\x,0) rectangle ++(\w,0.95);
    \node at (\x+1.15,0.48) {\rotulo};
  }
  \draw[->,thick] (0,-0.28) -- (11.75,-0.28) node[right] {evid\^encia};
\end{tikzpicture}
\end{center}

\begin{itemize}
  \item Os cortes v\^em de faixas de $p$-valor da distribui\c{c}\~ao te\'orica;
  \item o HMM recebe a sequ\^encia de labels, n\~ao o EEG bruto;
  \item frequ\^encias laterais s\~ao preservadas como controle do detector.
\end{itemize}
\end{frame}

% ====================================================================
\begin{frame}{5.1 Como os labels s\~ao obtidos}
\small

\begin{center}
  \includegraphics[width=0.94\textwidth]{figs/ord_distribuicao_nula.png}
\end{center}

\vspace{-0.25cm}
Para cada MSC observada, calculamos a \textbf{\'area de cauda} sob $H_0$:
\[
  p=P_{H_0}(\mathrm{MSC}\geq \mathrm{MSC}_{obs}).
\]
O intervalo que cont\'em esse $p$-valor determina o label emitido para o HMM.
Valores mais \`a direita s\~ao menos prov\'aveis sob $H_0$ e recebem labels de
maior evid\^encia.
\end{frame}

\begin{frame}{7. Como as tabelas s\~ao constru\'idas}
\normalsize
\begin{columns}[T]
\column{0.49\textwidth}
\begin{block}{Matriz de emiss\~ao $B$}
\begin{itemize}
  \item \textbf{Ausente}: histogramas de labels em grava\c{c}\~oes sem est\'imulo;
  \item \textbf{Presente}: os mesmos sinais com uma senoide coerente de pequena
  amplitude, apenas como cen\'ario sint\'etico;
  \item as contagens das frequ\^encias s\~ao reunidas e normalizadas.
\end{itemize}
\end{block}

\column{0.49\textwidth}
\begin{block}{Matriz de transi\c{c}\~ao $A$}
\begin{itemize}
  \item Cada grava\c{c}\~ao contribui com perman\^encias dentro de seu estado;
  \item uma sa\'ida impl\'icita evita perman\^encia igual a 100\%;
  \item a tabela expressa dura\c{c}\~ao induzida pelas grava\c{c}\~oes e pelo
  janelamento.
\end{itemize}
\end{block}
\end{columns}

\begin{alertblock}{Interpreta\c{c}\~ao correta}
Esta parametriza\c{c}\~ao \`e explorat\'oria: n\~ao \`e Baum--Welch, n\~ao representa
transi\c{c}\~oes fisiol\'ogicas observadas e n\~ao produz probabilidades cl\'inicas
validadas.
\end{alertblock}
\end{frame}

\begin{frame}{8. Da sequ\^encia ao caminho mais prov\'avel}
\large
Para cada frequ\^encia, o algoritmo de Viterbi combina a observa\c{c}\~ao atual
com as probabilidades de emiss\~ao e transi\c{c}\~ao.

\begin{center}
\begin{tikzpicture}[x=2.2cm,y=1.35cm,every node/.style={font=\scriptsize}]
  \foreach \t/\n/\obs in {0/1/baixo,1/2/m\'edio,2/3/alto,3/4/alto} {
    \node[state,minimum size=1.02cm,fill=colAbsent!40] (A\t) at (\t,1)
      {Ausente};
    \node[state,minimum size=1.02cm,fill=colPresent!45] (P\t) at (\t,0)
      {Presente};
    \node at (\t,-0.82) {$o_{\n}$: \obs};
  }
  \foreach \t/\u in {0/1,1/2,2/3} {
    \draw[->,thin,gray] (A\t) -- (A\u);
    \draw[->,thin,gray] (A\t) -- (P\u);
    \draw[->,thin,gray] (P\t) -- (A\u);
    \draw[->,thin,gray] (P\t) -- (P\u);
  }
  \draw[arrow,colSignal,line width=1.3pt] (A0) -- (A1);
  \draw[arrow,colSignal,line width=1.3pt] (A1) -- (P2);
  \draw[arrow,colSignal,line width=1.3pt] (P2) -- (P3);
\end{tikzpicture}
\end{center}

O caminho destacado \`e somente uma \textbf{ilustra\c{c}\~ao}. A infer\^encia \`e
feita separadamente para cada sequ\^encia; participantes, condi\c{c}\~oes e
frequ\^encias n\~ao s\~ao concatenados.
\end{frame}

\vspace{0.4cm}
\Large
\begin{itemize}
  \item A FFT isola a componente de interesse;
  \item a ORD mede nível de evidência em cada janela;
  \item o teste estat\'istico transforma a medida em evid\^encia ordinal;
  \item o HMM organiza essa evid\^encia ao longo do tempo.
\end{itemize}

\begin{frame}{10. Mensagem final}
\Large
\begin{block}{Ideia central}
O detector espectral fornece evid\^encia local; o HMM funciona como uma camada
temporal que favorece sequ\^encias coerentes de estados.
\end{block}

\begin{itemize}
  \item A contribui\c{c}\~ao para a disciplina est\'a no janelamento, FFT,
  coer\^encia e teste estat\'istico;
  \item o HMM \`e apresentado como uma aplica\c{c}\~ao simples dessas medidas;
  \item a configura\c{c}\~ao atual serve para estudo metodol\'ogico;
  \item qualquer uso cl\'inico exigiria calibra\c{c}\~ao e valida\c{c}\~ao
  independentes.
\end{itemize}

\vfill
\centering
\large Nenhum valor desta apresenta\c{c}\~ao deve ser interpretado como
resultado cl\'inico.
\end{frame}
